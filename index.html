// Configuração do Firebase (Substitua pelos dados do seu painel Firebase) const firebaseConfig = { apiKey: "SUA_API_KEY", authDomain: "seu-projeto.firebaseapp.com", databaseURL: "https://seu-projeto-default-rtdb.firebaseio.com", projectId: "seu-projeto", storageBucket: "seu-projeto.appspot.com", messagingSenderId: "SUA_ID", appId: "SUA_APP_ID" }; // Inicializar o Firebase firebase.initializeApp(firebaseConfig); const database = firebase.database(); // Fases do seu Fluxograma const FASES_PRODUCAO = [ "Customer care", "Aprovações fusion", "Plano de produção", "Estoque", "Movimentação expedição", "Pesagem", "Delivery", "Customer feedback" ]; // FUNÇÃO: Criar ou Atualizar Amostra por Código de Barras function registrarBipeCódigoBarras(codigoEtiqueta, modelo, observacao) { const agora = Date.now(); const refAmostra = database.ref('amostras/' + codigoEtiqueta); refAmostra.once('value').then((snapshot) => { if (!snapshot.exists()) { // Se o compressor nunca foi bipado, inicia na Fase 1 (Customer Care) const novaAmostra = { codigoBarras: codigoEtiqueta, modelo: modelo || "Compressor Padrão", faseAtualIndex: 0, criadoEm: agora, historico: { 0: { fase: FASES_PRODUCAO[0], chegouEm: agora, concluidoEm: null, obs: observacao || "Entrada inicial por código de barras." } } }; refAmostra.set(novaAmostra); } else { // Se já existe, avança para a próxima fase const dados = snapshot.val(); const indexAtual = dados.faseAtualIndex; if (indexAtual >= FASES_PRODUCAO.length - 1) { alert("Este compressor já concluiu todo o fluxo!"); return; } const novoIndex = indexAtual + 1; const atualizacao = {}; // Fecha a fase anterior calculando o timestamp de saída atualizacao[`faseAtualIndex`] = novoIndex; atualizacao[`historico/${indexAtual}/concluidoEm`] = agora; atualizacao[`historico/${indexAtual}/obs`] = observacao || "Passagem limpa."; // Abre o cronómetro da nova fase atualizacao[`historico/${novoIndex}`] = { fase: FASES_PRODUCAO[novoIndex], chegouEm: agora, concluidoEm: null, obs: "" }; refAmostra.update(atualizacao); } }); } // FUNÇÃO: Calcular o Delay Exato (Gargalo) function calcularDelayFase(chegouEm, concluidoEm) { const fim = concluidoEm ? concluidoEm : Date.now(); const diferencaMs = fim - chegouEm; const minutos totais = Math.round(diferencaMs / 60000); if (minutosTotais < 60) { return `${minutosTotais} min`; } else { const horas = Math.floor(minutosTotais / 60); const minsRestantes = minutosTotais % 60; return `${horas}h e ${minsRestantes}min`; } }
